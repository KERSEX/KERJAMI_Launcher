cmake_minimum_required(VERSION 3.24)

project(CommunityLauncher
    VERSION 0.1.0
    DESCRIPTION "Portable launcher for approved community projects"
    LANGUAGES CXX RC)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT WIN32)
    message(FATAL_ERROR "The launcher targets Windows 10/11 x64 only.")
endif()

include(cmake/Branding.cmake)

# ZlibPrivate is the zlib that is built into Qt. The ZIP reader uses it to
# inflate archive entries, so no archive library has to be added (D-27).
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Network Widgets ZlibPrivate)
qt_standard_project_setup()

# Everything runnable lands in one folder, which doubles as the portable root
# during development (settings, logs and the library are created next to the EXE).
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

set(LAUNCHER_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
configure_file(src/app/BrandingConfig.h.in "${LAUNCHER_GENERATED_DIR}/BrandingConfig.h" @ONLY)

# Shared compiler settings for all first-party targets.
function(launcher_target_defaults target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    target_compile_definitions(${target} PRIVATE
        QT_NO_CAST_FROM_ASCII
        QT_NO_CAST_TO_ASCII
        QT_NO_URL_CAST_FROM_STRING
        QT_NO_FOREACH
        QT_DISABLE_DEPRECATED_UP_TO=0x060500)
endfunction()

# Executables must not depend on compiler runtime DLLs (libc++, libunwind):
# the portable folder has to run on a clean Windows installation.
function(launcher_executable_defaults target)
    launcher_target_defaults(${target})
    if(MINGW)
        target_link_options(${target} PRIVATE -static)
    endif()
endfunction()

# --- Launcher executable ------------------------------------------------------
# Icon and version information of the EXE (shown by Explorer and the taskbar).

configure_file(src/app/launcher.rc.in "${LAUNCHER_GENERATED_DIR}/launcher.rc" @ONLY)
set_source_files_properties("${LAUNCHER_GENERATED_DIR}/launcher.rc"
    PROPERTIES OBJECT_DEPENDS "${LAUNCHER_ICON_FILE}")

qt_add_executable(launcher WIN32
    src/main.cpp
    src/app/Application.cpp
    src/app/Application.h
    "${LAUNCHER_GENERATED_DIR}/launcher.rc"
)
